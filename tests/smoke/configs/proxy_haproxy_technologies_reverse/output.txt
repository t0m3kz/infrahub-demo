#
# Device:     DC3-PRX-01
# Proxy type: reverse
# HA mode:    active-passive
#

global
    log stdout local0
    chroot /var/lib/haproxy
    stats socket /run/haproxy/admin.sock mode 660 level admin expose-fd listeners
    stats timeout 30s
    user haproxy
    group haproxy
    daemon
    maxconn 50000

    # SSL defaults
    ca-base /etc/ssl/certs
    crt-base /etc/ssl/private
    ssl-default-bind-ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384
    ssl-default-bind-options ssl-min-ver TLSv1.2 no-tls-tickets

defaults
    log     global
    mode    http
    option  httplog
    option  dontlognull
    option  forwardfor
    option  http-server-close
    retries 3
    timeout connect  5s
    timeout client   30s
    timeout server   30s
    timeout tunnel   1h
    maxconn 3000

#
# --- Interfaces ---
# mgmt: OOB management — 192.168.1.1/24
# eth0: Ingress uplink
# eth1: Egress uplink
#
# --- Stats ---
frontend stats
    bind 192.168.1.1:8404
    mode http
    stats enable
    stats uri /stats
    stats refresh 10s
    stats admin if TRUE
    no log
#
# --- Reverse proxy ---
frontend reverse_frontend
    bind *:80
    bind *:443 ssl crt /etc/ssl/private/server.pem
    mode http
    option httplog
    http-request redirect scheme https unless { ssl_fc }
    default_backend reverse_backend

backend reverse_backend
    mode http
    balance roundrobin
    option httpchk GET /health
    server-template upstream 10 0.0.0.0:0 disabled check
#
# --- Peers (HA state sync) ---
peers haproxy_cluster
    peer DC3-PRX-01 DC3-PRX-01:1024
    peer DC3-PRX-02 DC3-PRX-02:1024
