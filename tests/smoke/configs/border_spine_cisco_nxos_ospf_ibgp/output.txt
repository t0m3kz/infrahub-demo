!
! dc1-border-spine-01 Configuration
!
hostname dc1-border-spine-01
!
feature ospf
feature bgp
feature interface-vlan
feature lacp
feature vpc
feature lldp
!

! VLANs
vlan 100
  name Customer-A
!
vlan 200
  name Customer-A
!
!
! Interfaces
interface Ethernet1
  description to spine-01
  no shutdown
  ip address 10.1.0.1/31

  ip ospf area 0.0.0.0
!
interface Ethernet2
  description to spine-02
  no shutdown
  ip address 10.1.0.3/31

  ip ospf area 0.0.0.0
!
interface Ethernet10
  description to server-01
  no shutdown

  switchport
  switchport mode access
  switchport access vlan 100
!
interface Loopback0
  description Router ID
  no shutdown
  ip address 10.0.0.1/32

!
!
! OSPF Configuration

router ospf 1
  router-id 10.0.0.1
  auto-cost reference-bandwidth 10000
  passive-interface default
  no passive-interface Ethernet1
  no passive-interface Ethernet2
!
interface Ethernet1
  ip router ospf 1 area 0.0.0.0
interface Ethernet2
  ip router ospf 1 area 0.0.0.0

!
! BGP Configuration

router bgp 65000
  router-id 10.0.0.1
  bgp cluster-id 10.0.0.1
  graceful-restart
  !
  template peer-policy EVPN-PEERS-POLICY
    send-community
    route-reflector-client
    next-hop-unchanged
  !
  template peer-session EVPN-PEERS-SESSION
    update-source loopback0
  !
  address-family ipv4 unicast
    redistribute direct route-map RM-CONN-2-BGP
    maximum-paths 8
  !
  address-family l2vpn evpn
  !
  neighbor 10.0.0.100
    remote-as 65000
    description spine-01
    inherit peer-session EVPN-PEERS-SESSION
    address-family l2vpn evpn
      inherit peer-policy EVPN-PEERS-POLICY
    !
  neighbor 10.0.0.101
    remote-as 65000
    description spine-02
    inherit peer-session EVPN-PEERS-SESSION
    address-family l2vpn evpn
      inherit peer-policy EVPN-PEERS-POLICY
    !
  vrf VRF_A
    address-family ipv4 unicast
      advertise l2vpn evpn
      redistribute direct route-map RM-VRF-DIRECT-2-EVPN
  !
!

!
!
! VXLAN Configuration (netlab-style multivendor) — border-leaf is a VTEP too,
! terminating customer/shared VNIs the same as a regular leaf.
!
feature nv overlay
feature vn-segment-vlan-based
feature nve
!
!
! VRF Contexts (symmetric IRB — one L3 VNI per tenant)
!
vrf context VRF_A
  vni 50001
  rd 10.0.0.1:50001
  address-family ipv4 unicast
    route-target both 65000:50001
    route-target both 65000:50001 evpn
!
interface nve1
  source-interface Loopback0
  host-reachability protocol bgp
  member vni 10100
    suppress-arp
    ingress-replication protocol bgp
  member vni 50001 associate-vrf
  no shutdown
!
!
! EVPN Configuration
!
evpn
  vni 10100 l2
    rd 10.0.0.1:10100
    route-target import 65000:10100
    route-target export 65000:10100
!
!
! VLAN to VNI mapping
!
vlan 100
  vn-segment 10100
!
! L3 VNI Transit SVIs (symmetric IRB — no host addressing, forwarding only)
!
vlan 3900
  name L3VNI_VRF_A
  vn-segment 50001
!
interface Vlan3900
  description L3VNI transit SVI for VRF VRF_A
  vrf member VRF_A
  ip forward
  mtu 9216
  no shutdown
!
!
! Route-maps
route-map RM-CONN-2-BGP permit 10
  match interface loopback0
!
route-map RM-VRF-DIRECT-2-EVPN permit 10
!
! Anycast Gateway — identical MAC on every leaf/border-leaf in the fabric
fabric forwarding anycast-gateway-mac 001c7300dc01
!
! VLAN SVI Configuration
!
interface Vlan100
  description Customer-A
  vrf member VRF_A
  ip address 10.100.0.1/24
  fabric forwarding mode anycast-gateway
  no shutdown
!
interface Vlan200
  description Customer-A
  ip address 10.200.0.1/24
  fabric forwarding mode anycast-gateway
  no shutdown
!
!

line console
line vty
