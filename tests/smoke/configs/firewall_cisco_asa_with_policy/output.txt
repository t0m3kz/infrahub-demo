!
hostname dc1-fw-01
!
! --- Interfaces ---
interface ethernet1/1.10
 vlan 10
 ip address 10.0.1.1 255.255.255.252
 nameif internal
 security-level 100
 description Internal LAN link
 no shutdown
!
interface ethernet1/1.20
 vlan 20
 ip address 10.0.2.1 255.255.255.252
 nameif dmz
 security-level 50
 description DMZ link
 no shutdown
!
interface ethernet1/1.30
 vlan 30
 ip address 10.0.3.1 255.255.255.252
 nameif external
 security-level 0
 description External link
 no shutdown
!
! --- Static Routes (return path to leaf per zone) ---
route dmz 10.0.2.0 255.255.255.0 10.0.2.2 1
route internal 10.0.1.0 255.255.255.0 10.0.1.2 1
route internal 10.0.10.0 255.255.255.0 10.0.1.2 1
!
! --- Access Control Lists ---
access-list dmz-in extended permit tcp 10.0.2.0 255.255.255.0 10.0.1.0 255.255.255.0 eq 443 log
access-list dmz-in extended permit tcp 10.0.2.0 255.255.255.0 10.0.1.0 255.255.255.0 eq 22 log
access-list internal-in extended permit ip 10.0.1.0 255.255.255.0 10.0.2.0 255.255.255.0
access-list global-in extended deny ip any any log
!
! --- Access Groups ---
access-group internal-in in interface internal
access-group dmz-in in interface dmz
access-group external-in in interface external
end
