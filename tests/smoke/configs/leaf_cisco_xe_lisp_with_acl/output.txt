! LISP Configuration (Catalyst 9000, IOS-XE)
!
! Tenant VRFs (the EID tables of the L3 instance-ids)
! unverified-no-lab: a plain `vrf definition` with an IPv4 address family is all the EID table and the SVIs need
vrf definition VRF_A
 address-family ipv4
 exit-address-family
!
! unverified-no-lab: `router lisp` and everything under it
router lisp
 ! unverified-no-lab: locator-set RLOC on the underlay loopback, used as the default locator and as the source locator
 locator-table default
 locator-set RLOC
  IPv4-interface Loopback0 priority 10 weight 10
  exit-locator-set
 locator default-set RLOC
 ipv4 source-locator Loopback0
 ! unverified-no-lab: service ipv4: ITR and ETR toward every map-server, VXLAN encapsulation
 service ipv4
  encapsulation vxlan
  itr map-resolver 10.255.0.1
  itr map-resolver 10.255.0.2
  etr map-server 10.255.0.1 key s3cr3t
  etr map-server 10.255.0.1 proxy-reply
  etr map-server 10.255.0.2 key s3cr3t
  etr map-server 10.255.0.2 proxy-reply
  itr
  etr
  exit-service-ipv4
 ! unverified-no-lab: service ethernet: ITR and ETR toward every map-server
 service ethernet
  itr map-resolver 10.255.0.1
  itr map-resolver 10.255.0.2
  etr map-server 10.255.0.1 key s3cr3t
  etr map-server 10.255.0.1 proxy-reply
  etr map-server 10.255.0.2 key s3cr3t
  etr map-server 10.255.0.2 proxy-reply
  itr
  etr
  exit-service-ethernet
 ! unverified-no-lab: L3 instance-id 50001 for VRF VRF_A: dynamic-eid with the gateway subnets, IPv4 EID table, default map-request
 instance-id 50001
  dynamic-eid VRF_A-IPV4
   database-mapping 10.100.0.0/24 locator-set RLOC
   exit-dynamic-eid
  service ipv4
   eid-table vrf VRF_A
   map-cache 0.0.0.0/0 map-request
   exit-service-ipv4
  exit-instance-id
 ! unverified-no-lab: L2 instance-id 10100 for VLAN 100: Ethernet EID table and the local MAC database mapping
 instance-id 10100
  service ethernet
   eid-table vlan 100
   database-mapping mac locator-set RLOC
   exit-service-ethernet
  exit-instance-id
