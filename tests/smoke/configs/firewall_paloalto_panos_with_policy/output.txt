# --- Interfaces & Zone Assignments ---
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.10 tag 10
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.10 ipv4 addr 10.0.1.1/30
set network interface ethernet ethernet1/1.10 comment "Internal LAN link"
set zone PROD network layer3 ethernet1/1.10
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.20 tag 20
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.20 ipv4 addr 10.0.2.1/30
set network interface ethernet ethernet1/1.20 comment "DMZ link"
set zone NONPROD network layer3 ethernet1/1.20
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.30 tag 30
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.30 ipv4 addr 10.0.3.1/30
set network interface ethernet ethernet1/1.30 comment "External link"
set zone INTERNET network layer3 ethernet1/1.30
#
#
# --- Address Objects (one per zone member CIDR) ---
set address addr-10-0-1-0-24 ip-netmask 10.0.1.0/24
set address addr-10-0-1-0-24 description "Zone: PROD"
set address addr-10-0-10-0-24 ip-netmask 10.0.10.0/24
set address addr-10-0-10-0-24 description "Zone: PROD"
set address addr-10-0-2-0-24 ip-netmask 10.0.2.0/24
set address addr-10-0-2-0-24 description "Zone: NONPROD"
#
# --- Address Groups (one per zone) ---
set address-group ag-PROD static [ addr-10-0-1-0-24 addr-10-0-10-0-24 ]
set address-group ag-NONPROD static [ addr-10-0-2-0-24 ]
#
# --- Virtual router interfaces ---
set network virtual-router default interface ethernet1/1.10
set network virtual-router default interface ethernet1/1.20
set network virtual-router default interface ethernet1/1.30
#
# --- Static Routes (return path to leaf per zone) ---
set network virtual-router default routing-table ip static-route NONPROD-1 destination 10.0.2.0/24
set network virtual-router default routing-table ip static-route NONPROD-1 nexthop ip-address 10.0.2.2
set network virtual-router default routing-table ip static-route NONPROD-1 interface ethernet1/1.20
set network virtual-router default routing-table ip static-route PROD-2 destination 10.0.1.0/24
set network virtual-router default routing-table ip static-route PROD-2 nexthop ip-address 10.0.1.2
set network virtual-router default routing-table ip static-route PROD-2 interface ethernet1/1.10
set network virtual-router default routing-table ip static-route PROD-3 destination 10.0.10.0/24
set network virtual-router default routing-table ip static-route PROD-3 nexthop ip-address 10.0.1.2
set network virtual-router default routing-table ip static-route PROD-3 interface ethernet1/1.10
#
#
# --- Firewall Contexts (VDOM/vsys) — one per tenant, per ManagedFirewallContext ---
#
set vsys shared-dc1-fw-01-policy

# --- Security Policies in vsys shared-dc1-fw-01-policy ---

set vsys shared-dc1-fw-01-policy rulebase security rules allow-https to PROD

set vsys shared-dc1-fw-01-policy rulebase security rules allow-https service svc-tcp-443
set vsys shared-dc1-fw-01-policy rulebase security rules allow-https action allow
set vsys shared-dc1-fw-01-policy rulebase security rules allow-https profile-setting group strict-av
set vsys shared-dc1-fw-01-policy rulebase security rules allow-https log-end yes
set vsys shared-dc1-fw-01-policy rulebase security rules allow-https description "Allow HTTPS from DMZ to Internal"
set vsys shared-dc1-fw-01-policy rulebase security rules allow-ssh from NONPROD
set vsys shared-dc1-fw-01-policy rulebase security rules allow-ssh to PROD

set vsys shared-dc1-fw-01-policy rulebase security rules allow-ssh log-end yes
set vsys shared-dc1-fw-01-policy rulebase security rules allow-ssh description "Allow SSH from DMZ to Internal"

set vsys shared-dc1-fw-01-policy rulebase security rules allow-any to NONPROD
