! Hostname configuration
hostname dc1-leaf-01

! Enable required features
nv overlay evpn
feature ospf
feature bgp
feature interface-vlan
feature vn-segment-vlan-based
feature nv overlay
vlan 100
  name Customer-A
  vn-segment 100

vlan 200
  name Customer-A
  vn-segment 200

interface Ethernet1
  description to spine-01
  ip address 10.1.0.1/31
  no shutdown
!

interface Ethernet2
  description to spine-02
  ip address 10.1.0.3/31
  no shutdown
!

interface Ethernet10
  description to server-01
  switchport
  switchport mode access
  switchport access vlan 100
  no shutdown
!

interface Loopback0
  description Router ID
  ip address 10.0.0.1/32
  no shutdown
!

router bgp 65001
  router-id 10.0.0.1
  no bgp default ipv4-unicast
  distance bgp 20 200 200
  bgp graceful-restart
  maximum-paths 4 ecmp 64
  !
  neighbor EVPN-OVERLAY peer-group
  neighbor EVPN-OVERLAY update-source Loopback0
  neighbor EVPN-OVERLAY send-community
  neighbor EVPN-OVERLAY send-community extended
  neighbor EVPN-OVERLAY ebgp-multihop 255
  !
  neighbor UNDERLAY-PEERS peer-group
  neighbor UNDERLAY-PEERS send-community
  neighbor UNDERLAY-PEERS bfd
  neighbor 10.1.0.0 peer-group UNDERLAY-PEERS
  neighbor 10.1.0.0 remote-as 65100
  neighbor 10.1.0.0 description spine-01
  neighbor 10.1.0.2 peer-group UNDERLAY-PEERS
  neighbor 10.1.0.2 remote-as 65101
  neighbor 10.1.0.2 description spine-02
  neighbor 10.0.0.100 peer-group EVPN-OVERLAY
  neighbor 10.0.0.100 remote-as 65100
  neighbor 10.0.0.100 description spine-01
  neighbor 10.0.0.101 peer-group EVPN-OVERLAY
  neighbor 10.0.0.101 remote-as 65101
  neighbor 10.0.0.101 description spine-02
  !
  address-family ipv4
    redistribute connected route-map RM-CONN-2-BGP
    neighbor UNDERLAY-PEERS activate
  !
  address-family evpn
    neighbor EVPN-OVERLAY activate
!


!
! VXLAN Configuration with Microsegmentation (VRF-aware VNIs)
!
! Microsegmentation enabled: 1 VRF(s) for tenant isolation
!
interface vtep
  source Loopback0
  !
  ! L2 VNI Mappings (VLAN segments)
  map vni 10100 vlan 100
  !
  ! L3 VNI Mappings (VRF segments for microsegmentation)
  map vni 50001 vrf VRF_A
  no shutdown
!
!
! VRF Definitions (Microsegmentation)
!
vrf VRF_A
  description Tenant VRF for microsegmentation
!
!
! EVPN Configuration (FRR)
!
router bgp 65001
  address-family l2vpn evpn
    advertise-all-vni
    !
    ! L2 VNI EVPN Configuration
    vni 10100
      rd 10.0.0.1:10100
      route-target import 65000:10100
      route-target export 65000:10100
    !
    ! L3 VNI EVPN Configuration (VRF-aware)
    vni 50001
      rd 10.0.0.1:50001
      route-target import 65000:50001
      route-target export 65000:50001
  exit-address-family
!

! Configure route-maps
route-map RM-CONN-2-BGP permit 10
  match interface Loopback0
!
! VLAN SVI Configuration
!
interface Vlan100
  description Customer-A
  vrf forwarding VRF_A
  ip address 10.100.0.1/24
  no shutdown
!
interface Vlan200
  description Customer-A
  ip address 10.200.0.1/24
  no shutdown
!

! End of configuration
