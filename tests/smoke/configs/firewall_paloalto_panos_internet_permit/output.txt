set deviceconfig system hostname dc1-fw-01

# --- Interfaces & Zone Assignments ---
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.10 tag 10
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.10 ipv4 addr 10.0.1.1/30
set network interface ethernet ethernet1/1.10 comment "Internal LAN link"
set zone internal network layer3 ethernet1/1.10
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.30 tag 30
set network interface ethernet ethernet1/1 layer3 units ethernet1/1.30 ipv4 addr 10.0.3.1/30
set network interface ethernet ethernet1/1.30 comment "Internet link"
set zone INTERNET-ZONE network layer3 ethernet1/1.30
#
#
# --- Address Objects (one per zone member CIDR) ---
set address addr-10-0-1-0-24 ip-netmask 10.0.1.0/24
set address addr-10-0-1-0-24 description "Zone: internal"
#
# --- Address Groups (one per zone) ---
set address-group ag-internal static [ addr-10-0-1-0-24 ]
set address-group ag-INTERNET-ZONE static [ any ]
#
# --- Virtual router interfaces ---
set network virtual-router default interface ethernet1/1.10
set network virtual-router default interface ethernet1/1.30
#
# --- Static Routes (return path to leaf per zone) ---
set network virtual-router default routing-table ip static-route VRF-INTERNAL-1 destination 10.0.1.0/24
set network virtual-router default routing-table ip static-route VRF-INTERNAL-1 nexthop ip-address 10.0.1.2
set network virtual-router default routing-table ip static-route VRF-INTERNAL-1 interface ethernet1/1.10
#
#
# --- Security Policies ---
set rulebase security rules allow-internet-https from internal
set rulebase security rules allow-internet-https to INTERNET-ZONE
set rulebase security rules allow-internet-https source 10.0.1.0/24
set rulebase security rules allow-internet-https destination any
set rulebase security rules allow-internet-https application any
set service svc-tcp-443 protocol tcp port 443
set rulebase security rules allow-internet-https service svc-tcp-443
set rulebase security rules allow-internet-https action allow
set rulebase security rules allow-internet-https description "Allow HTTPS from Internal to the Internet"
set rulebase security rules implicit-deny-all from any
set rulebase security rules implicit-deny-all to any
set rulebase security rules implicit-deny-all source any
set rulebase security rules implicit-deny-all destination any
set rulebase security rules implicit-deny-all application any
set rulebase security rules implicit-deny-all service any
set rulebase security rules implicit-deny-all action deny
set rulebase security rules implicit-deny-all log-end yes
set rulebase security rules implicit-deny-all description "Implicit deny all"
